Task 1

The menu skeleton

Start with the loop only. Each option just prints a placeholder for now.

powershell · admin-tool.ps1
function Show-Menu {
    Write-Host '===== Admin Tool =====' -ForegroundColor Cyan
    Write-Host '1) System info'
    Write-Host '2) List big files'
    Write-Host '3) Log summary'
    Write-Host '4) Student report'
    Write-Host '5) Exit'
}

do {
    Show-Menu
    $choice = Read-Host 'Choose 1-5'
    switch ($choice) {
        '1'     { Write-Host 'System info: coming soon' }
        '2'     { Write-Host 'Big files: coming soon' }
        '3'     { Write-Host 'Log summary: coming soon' }
        '4'     { Write-Host 'Student report: coming soon' }
        '5'     { Write-Host 'Goodbye!' }
        default { Write-Host "'$choice' is not a valid choice." -ForegroundColor Yellow }
    }
} until ($choice -eq '5')

Task 2

Options 1 and 2: system info and big files

Add these functions above the do loop, and call them from cases '1' and '2'.

powershell · admin-tool.ps1 (add)
function Show-SystemInfo {
    [pscustomobject]@{
        Computer   = [Environment]::MachineName
        User       = [Environment]::UserName
        OS         = $PSVersionTable.OS
        PowerShell = $PSVersionTable.PSVersion.ToString()
        Time       = Get-Date -Format 'yyyy-MM-dd HH:mm'
    } | Format-List | Out-Host
}

function Show-BigFiles {
    param([string]$Path = '.', [int]$Top = 5)
    Get-ChildItem -Path $Path -File -Recurse |
        Sort-Object Length -Descending |
        Select-Object -First $Top Name,
            @{ Name = 'SizeKB'; Expression = { [math]::Round($_.Length / 1KB, 1) } } |
        Format-Table -AutoSize | Out-Host
}

Task 3

Option 3: log summary

powershell · admin-tool.ps1 (add)
function Show-LogSummary {
    param([string]$Path = 'datasets/app.log')
    $lines = Get-Content -Path $Path
    $lines | ForEach-Object { ($_ -split '\s+')[2] } |
        Group-Object -NoElement |
        Select-Object Name, Count |
        Format-Table -AutoSize | Out-Host
    Write-Host 'Error lines:' -ForegroundColor Red
    $lines | Where-Object { $_ -match ' ERROR ' } | ForEach-Object { Write-Host "  $_" }
}

Task 4

Option 4: student report

A student passes only if every subject is 40 or more.

powershell · admin-tool.ps1 (add)
function Show-StudentReport {
    param([string]$Path = 'datasets/students.csv')
    $rows = Import-Csv -Path $Path | ForEach-Object {
        $m = [int]$_.Maths; $s = [int]$_.Science; $e = [int]$_.English
        if ($m -ge 40 -and $s -ge 40 -and $e -ge 40) { $result = 'Pass' } else { $result = 'Fail' }
        [pscustomobject]@{
            Name    = $_.Name
            Average = [math]::Round(($m + $s + $e) / 3, 1)
            Result  = $result
        }
    }
    $rows | Sort-Object Average -Descending | Format-Table -AutoSize | Out-Host
    $passed = ($rows | Where-Object Result -eq 'Pass').Count
    Write-Host "Passed: $passed of $($rows.Count)"
}

Full script

admin-tool.ps1 (complete)
powershell · admin-tool.ps1
<#
  admin-tool.ps1: a menu-driven admin tool (Unit 5 mini project)
  Run from the unit5 folder:  ./admin-tool.ps1
#>
param(
    [string]$LogPath     = 'datasets/app.log',
    [string]$StudentPath = 'datasets/students.csv',
    [string]$FolderPath  = '.'
)

function Show-Menu {
    Write-Host '===== Admin Tool =====' -ForegroundColor Cyan
    Write-Host '1) System info'
    Write-Host '2) List big files'
    Write-Host '3) Log summary'
    Write-Host '4) Student report'
    Write-Host '5) Exit'
}

function Show-SystemInfo {
    [pscustomobject]@{
        Computer   = [Environment]::MachineName
        User       = [Environment]::UserName
        OS         = $PSVersionTable.OS
        PowerShell = $PSVersionTable.PSVersion.ToString()
        Time       = Get-Date -Format 'yyyy-MM-dd HH:mm'
    } | Format-List | Out-Host
}

function Show-BigFiles {
    param([string]$Path = '.', [int]$Top = 5)
    try {
        Get-ChildItem -Path $Path -File -Recurse -ErrorAction Stop |
            Sort-Object Length -Descending |
            Select-Object -First $Top Name,
                @{ Name = 'SizeKB'; Expression = { [math]::Round($_.Length / 1KB, 1) } } |
            Format-Table -AutoSize | Out-Host
    }
    catch {
        Write-Host "Cannot list files: $($_.Exception.Message)" -ForegroundColor Red
    }
}

function Show-LogSummary {
    param([string]$Path = 'datasets/app.log')
    try {
        $lines = Get-Content -Path $Path -ErrorAction Stop
        $lines | ForEach-Object { ($_ -split '\s+')[2] } |
            Group-Object -NoElement |
            Select-Object Name, Count |
            Format-Table -AutoSize | Out-Host
        Write-Host 'Error lines:' -ForegroundColor Red
        $lines | Where-Object { $_ -match ' ERROR ' } | ForEach-Object { Write-Host "  $_" }
    }
    catch {
        Write-Host "Cannot read log: $($_.Exception.Message)" -ForegroundColor Red
    }
}

function Show-StudentReport {
    param([string]$Path = 'datasets/students.csv')
    try {
        $rows = Import-Csv -Path $Path -ErrorAction Stop | ForEach-Object {
            $m = [int]$_.Maths; $s = [int]$_.Science; $e = [int]$_.English
            if ($m -ge 40 -and $s -ge 40 -and $e -ge 40) { $result = 'Pass' } else { $result = 'Fail' }
            [pscustomobject]@{
                Name    = $_.Name
                Average = [math]::Round(($m + $s + $e) / 3, 1)
                Result  = $result
            }
        }
        $rows | Sort-Object Average -Descending | Format-Table -AutoSize | Out-Host
        $passed = ($rows | Where-Object Result -eq 'Pass').Count
        Write-Host "Passed: $passed of $($rows.Count)"
    }
    catch {
        Write-Host "Cannot read students: $($_.Exception.Message)" -ForegroundColor Red
    }
}

# ---------- main loop ----------
do {
    Show-Menu
    $choice = Read-Host 'Choose 1-5'
    switch ($choice) {
        '1'     { Show-SystemInfo }
        '2'     { Show-BigFiles -Path $FolderPath }
        '3'     { Show-LogSummary -Path $LogPath }
        '4'     { Show-StudentReport -Path $StudentPath }
        '5'     { Write-Host 'Goodbye!' }
        default { Write-Host "'$choice' is not a valid choice." -ForegroundColor Yellow }
    }
} until ($choice -eq '5')

Viva questions

Why is do/until a good fit for a menu?

The menu must appear at least once, and it should repeat until the user picks Exit. do/until runs first and tests after.

Why are the switch cases written as '1' and not 1?

Read-Host returns text. Quoting the cases makes it clear we compare text with text.

Why put each option in a function?

Each piece is short, has one job and can be tested on its own. The main loop stays easy to read.

What does the param() block at the top of a script do?

It lets the user pass values when running the script, such as ./admin-tool.ps1 -LogPath other.log. The defaults are used otherwise.
