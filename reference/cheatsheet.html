Script skeleton

bash · template.sh
#!/usr/bin/env bash
set -euo pipefail            # -e exit on error, -u unset var = error, pipefail: pipe fails if any part fails
IFS=$'\n\t'                  # optional: safer word splitting

readonly SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
log()  { printf '%s %s\n' "$(date +%T)" "$*" >&2; }
die()  { log "ERROR: $*"; exit 1; }
cleanup() { rm -f "${tmp:-}"; }
trap cleanup EXIT

[[ $# -ge 1 ]] || die "usage: $0 FILE"
tmp=$(mktemp)
main() { local file="$1"; log "processing $file"; }
main "$@"

Run: chmod +x template.sh && ./template.sh file · debug: bash -x script.sh · syntax check: bash -n script.sh · lint: shellcheck script.sh

Loops

bash
for f in *.log; do echo "$f"; done                  # glob
for i in {1..5}; do echo "$i"; done                 # brace range
for (( i=0; i<5; i++ )); do echo "$i"; done         # C-style
while read -r host ip _; do echo "$host=$ip"; done < servers.txt      # read a file line by line
while IFS=, read -r id name dept _; do echo "$name"; done < <(tail -n +2 employees.csv)
until ping -c1 -W1 "$h" &>/dev/null; do sleep 2; done                # retry until success
select opt in Start Stop Quit; do [[ $opt == Quit ]] && break; echo "$opt"; done   # menu
break / continue                                    # leave loop / next iteration

Functions and arrays

bash
greet() {                       # define
  local name="${1:-world}"      # local + default
  echo "hello $name"            # "return" data via stdout
  return 0                      # status 0-255 only
}
msg=$(greet Asha)               # capture output;  greet || echo failed  (use status)

arr=(web01 web02 "db 01")       # indexed array
echo "${arr[0]} ${#arr[@]}"     # first element, length
for h in "${arr[@]}"; do echo "$h"; done
arr+=(cache01)                  # append
declare -A port=([http]=80 [https]=443)     # associative array (Bash 4+)
for k in "${!port[@]}"; do echo "$k=${port[$k]}"; done
mapfile -t lines < file.txt     # file -> array, one line per element

ExamCron runs with a minimal environment (short PATH, no aliases, /bin/sh, home directory as cwd). Most "works in terminal, fails in cron" problems come from relative paths, missing PATH entries and unescaped %.
